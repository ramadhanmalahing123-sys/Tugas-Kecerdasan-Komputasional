#  FUNGSI DECISION TREE (IF-ELSE)
# ═══════════════════════════════════════════════════════════════
def tentukan_status(kehadiran, tugas):
    """
    Decision Tree sederhana:
      Kehadiran Tinggi → Aktif
        └─ Tugas Lengkap   → "Mahasiswa Disiplin"
        └─ Tugas Tdk Lengkap → "Perlu Tingkatkan Tugas"
      Kehadiran Rendah → Tidak Aktif
        └─ Tugas Lengkap   → "Rajin Tugas, Kurang Hadir"
        └─ Tugas Tdk Lengkap → "Perlu Perhatian Khusus"
    """
    if kehadiran == "Tinggi":
        status = "Aktif"
        if tugas == "Lengkap":
            keterangan = "Mahasiswa Disiplin ★"
            skor = 100
        else:
            keterangan = "Perlu Tingkatkan Tugas"
            skor = 70
    else:  # Rendah
        status = "Tidak Aktif"
        if tugas == "Lengkap":
            keterangan = "Rajin Tugas, Kurang Hadir"
            skor = 50
        else:
            keterangan = "Perlu Perhatian Khusus ⚠"
            skor = 20

    return status, keterangan, skor

def warna_status(status):
    if status == "Aktif":
        return f"{BOLD}{HIJAU}{status}{RESET}"
    return f"{BOLD}{MERAH}{status}{RESET}"


def warna_kehadiran(kehadiran):
    if kehadiran == "Tinggi":
        return f"{HIJAU}{kehadiran}{RESET}"
    return f"{KUNING}{kehadiran}{RESET}"


def warna_tugas(tugas):
    if tugas == "Lengkap":
        return f"{HIJAU}{tugas}{RESET}"
    return f"{ORANYE}{tugas}{RESET}"


def warna_keterangan(ket):
    if "Disiplin" in ket:
        return f"{BOLD}{CYAN}{ket}{RESET}"
    elif "Perhatian" in ket:
        return f"{BOLD}{MERAH}{ket}{RESET}"
    elif "Rajin" in ket:
        return f"{KUNING}{ket}{RESET}"
    return f"{DIM}{ket}{RESET}"

          f"{BOLD}{'SKOR':^7}{RESET}{BIRU}│{RESET}"
          f"{BOLD}{'KETERANGAN':^26}{RESET}{BIRU}│{RESET}")

        status, keterangan, skor = tentukan_status(mhs["kehadiran"], mhs["tugas"])

        # Simpan ke riwayat
        riwayat_log.append({

        tugas_str   = mhs["tugas"]

        nama_pad  = nama_str.center(12)
        had_pad   = had_str.center(12)

              f"{warna_keterangan(keterangan):} {BIRU}│{RESET}")

#  FITUR TAMBAHAN 1 : RINGKASAN STATISTIK
# ═══════════════════════════════════════════════════════════════
def tampilkan_statistik(data):
    total      = len(data)
    aktif      = sum(1 for m in data if tentukan_status(m["kehadiran"], m["tugas"])[0] == "Aktif")
    tidak_aktif = total - aktif
    disiplin   = sum(1 for m in data if "Disiplin" in tentukan_status(m["kehadiran"], m["tugas"])[1])
    perhatian  = sum(1 for m in data if "Perhatian" in tentukan_status(m["kehadiran"], m["tugas"])[1])

    rata_skor  = sum(tentukan_status(m["kehadiran"], m["tugas"])[2] for m in data) / total

    print(f"\n{BOLD}{CYAN}  📊  RINGKASAN STATISTIK KELAS{RESET}")
    print(f"  {'─'*40}")
    print(f"  Total Mahasiswa    : {BOLD}{PUTIH}{total}{RESET} orang")
    print(f"  Status Aktif       : {BOLD}{HIJAU}{aktif}{RESET} mahasiswa")
    print(f"  Status Tidak Aktif : {BOLD}{MERAH}{tidak_aktif}{RESET} mahasiswa")
    print(f"  Mahasiswa Disiplin : {BOLD}{CYAN}{disiplin}{RESET} mahasiswa ★")
    print(f"  Butuh Perhatian    : {BOLD}{ORANYE}{perhatian}{RESET} mahasiswa ⚠")
    print(f"  Rata-rata Skor     : {BOLD}{KUNING}{rata_skor:.1f}{RESET} / 100")

    # Bar chart sederhana
    print(f"\n  {DIM}Distribusi Status:{RESET}")
    bar_aktif   = "█" * aktif
    bar_tidak   = "█" * tidak_aktif
    print(f"  Aktif       {HIJAU}{bar_aktif:<20}{RESET} {aktif} mahasiswa")
    print(f"  Tidak Aktif {MERAH}{bar_tidak:<20}{RESET} {tidak_aktif} mahasiswa")


# ═══════════════════════════════════════════════════════════════
#  FITUR TAMBAHAN 2 : CARI MAHASISWA BERDASARKAN NAMA
# ═══════════════════════════════════════════════════════════════
def cari_mahasiswa(data):
    print(f"\n{BOLD}{MAGENTA}  🔍  PENCARIAN MAHASISWA{RESET}")
    print(f"  {'─'*40}")
    keyword = input(f"  Masukkan nama mahasiswa : ").strip().lower()

    hasil = [m for m in data if keyword in m["nama"].lower()]

    if not hasil:
        print(f"  {MERAH}Mahasiswa dengan nama '{keyword}' tidak ditemukan.{RESET}")
        return

    for mhs in hasil:
        status, keterangan, skor = tentukan_status(mhs["kehadiran"], mhs["tugas"])
        print(f"\n  {BG_ABU}{'─'*42}{RESET}")
        print(f"  {BOLD}NIM        :{RESET} {mhs['nim']}")
        print(f"  {BOLD}Nama       :{RESET} {MAGENTA}{mhs['nama']}{RESET}")
        print(f"  {BOLD}Kehadiran  :{RESET} {warna_kehadiran(mhs['kehadiran'])}")
        print(f"  {BOLD}Tugas      :{RESET} {warna_tugas(mhs['tugas'])}")
        print(f"  {BOLD}Status     :{RESET} {warna_status(status)}")
        print(f"  {BOLD}Skor       :{RESET} {KUNING}{skor}{RESET} / 100")
        print(f"  {BOLD}Keterangan :{RESET} {warna_keterangan(keterangan)}")
    print(f"  {BG_ABU}{'─'*42}{RESET}")

#  FITUR TAMBAHAN 3 : TAMBAH MAHASISWA BARU (SESI INI)

def tambah_mahasiswa(data):
    print(f"\n{BOLD}{HIJAU}  ➕  TAMBAH MAHASISWA BARU{RESET}")
    print(f"  {'─'*40}")

    nim  = input(f"  NIM Mahasiswa     : ").strip()
    nama = input(f"  Nama Mahasiswa    : ").strip()

    print(f"  Kehadiran  [1] Tinggi  [2] Rendah : ", end="")
    pilih_had = input().strip()
    kehadiran = "Tinggi" if pilih_had == "1" else "Rendah"

    print(f"  Tugas      [1] Lengkap [2] Tidak Lengkap : ", end="")
    pilih_tugas = input().strip()
    tugas = "Lengkap" if pilih_tugas == "1" else "Tidak Lengkap"

    mhs_baru = {"nim": nim, "nama": nama, "kehadiran": kehadiran, "tugas": tugas}
    data.append(mhs_baru)

    status, keterangan, skor = tentukan_status(kehadiran, tugas)
    print(f"\n  {HIJAU}✔ Mahasiswa {BOLD}{nama}{RESET}{HIJAU} berhasil ditambahkan!{RESET}")
    print(f"  Status     : {warna_status(status)}")
    print(f"  Keterangan : {warna_keterangan(keterangan)}")
    print(f"  Skor       : {KUNING}{skor}{RESET} / 100")
